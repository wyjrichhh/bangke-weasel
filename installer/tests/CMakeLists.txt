cmake_minimum_required(VERSION 3.21)
project(BangkeInstallerTests LANGUAGES CXX)
enable_testing()
add_executable(install_policy_test install_policy_test.cpp)
target_compile_features(install_policy_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(install_policy_test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(install_policy_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME install_policy COMMAND install_policy_test)
add_executable(model_files_test model_files_test.cpp)
target_compile_features(model_files_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(model_files_test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(model_files_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME model_files COMMAND model_files_test)

if(WIN32)
  add_executable(status_icon_test status_icon_test.cpp)
  target_compile_features(status_icon_test PRIVATE cxx_std_17)
  target_compile_options(status_icon_test PRIVATE /W4 /WX /utf-8)
  target_link_libraries(status_icon_test PRIVATE user32 gdi32 advapi32)
  add_test(NAME status_icon COMMAND status_icon_test)
endif()
